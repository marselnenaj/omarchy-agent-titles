cmake_minimum_required(VERSION 3.27)
project(agent-titles VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(BUILD_PLUGIN "Build the Hyprland decoration" ON)
find_package(nlohmann_json 3.11 REQUIRED)
if(BUILD_PLUGIN)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(deps REQUIRED IMPORTED_TARGET hyprland libdrm libinput libudev pixman-1 wayland-server xkbcommon glib-2.0)
  add_library(agent-titles SHARED native/main.cpp native/barDeco.cpp native/BarPassElement.cpp)
  target_compile_options(agent-titles PRIVATE -Wall -Wextra
    "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-fdebug-prefix-map=${CMAKE_SOURCE_DIR}=.")
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(agent-titles PRIVATE -fno-gnu-unique)
  endif()
  target_link_options(agent-titles PRIVATE -Wl,-z,now)
  target_link_libraries(agent-titles PRIVATE PkgConfig::deps nlohmann_json::nlohmann_json)
  set_target_properties(agent-titles PROPERTIES PREFIX "")
  install(TARGETS agent-titles LIBRARY DESTINATION lib)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(native-tests tests/native-tests.cpp)
  target_link_libraries(native-tests PRIVATE nlohmann_json::nlohmann_json)
  add_test(NAME native-model-and-settings COMMAND native-tests)
endif()
